JavaScript API

JavaScript API

كونترولنى باغلاش، ئىشلىتىش ۋە تازىلاش؛ SVG ھاسىل قىلىش ۋە سىنبەلگە ئىزدەش كۆرۈنمە يۈزى.

بۇ بەتتە
كىرگۈزۈش ۋە ھايات دەۋرىbindDialogbindDisclosure ۋە bindTabsvalidateForm ۋە bindFormshowToastSVG ۋە ئىزدەش APIپارامېتىر ۋە خاتالىق

كىرگۈزۈش ۋە ھايات دەۋرى

import {
  bindDialog, bindDisclosure, bindTabs, bindForm, validateForm, showToast,
  iconSVG, flagSVG, mapSVG, mapMetadata, iconNames, coloredIconNames, directionalIcons,
  iconCatalog, iconCategories, findIcons,
} from '@uyghurcommunity/ui';

يىلتىز import ھۆججىتى يۈكلەنگەندە DOM غا كىرمەيدۇ. كونترولنى باغلاشنى ئېلېمېنت قۇرۇلغاندىن كېيىن قىلىڭ. رامكا تەركىبى ياكى بەت پارچىسى ئۆچۈرۈلگەندە ئۇنىڭ destroy() نى چاقىرىڭ؛ يېڭى DOM غا يېڭى كونترول باغلاڭ.

bindDialog

const dialog = bindDialog(document.getElementById('settings'), {
  trigger: document.getElementById('open-settings'),
  initialFocus: document.getElementById('settings-title'),
});
dialog.open();
dialog.close('saved');
// When the owning view is removed:
dialog.destroy();

بىرىنچى پارامېتىر showModal() ۋە close() نى قوللايدىغان يەرلىك dialog بولۇشى كېرەك. ئۇنىڭغا aria-label ياكى aria-labelledby بېرىڭ. ئىچىدىكى data-ug-close كۇنۇپكىسى تاقايدۇ؛ قىممىتى native close نەتىجىسىگە ئۆتىدۇ. تاقىغاندا ئالدىنقى فوكۇس ئەگەر ھېلىمۇ بەتتە بولسا قايتۇرۇلىدۇ. بۇ كونترول يەرلىك dialog قوللىشىنىڭ ئورنىنى باسىدىغان polyfill ئەمەس.

bindDisclosure ۋە bindTabs

const menu = bindDisclosure(button, panel);
menu.setOpen(true);
menu.toggle();
menu.destroy();

const tabs = bindTabs(document.getElementById('tabs'));
tabs.select(document.getElementById('tab-two'), true);
tabs.destroy();

Disclosure غا button ۋە panel بېرىڭ، HTML دا aria-controls نى توغرا باغلاڭ. ئۇ expanded ۋە hidden ھالىتىنى باشقۇرىدۇ؛ ئىچكى مەزمۇندا Escape بېسىلسا يېپىلىدۇ.

Tabs قاچىسى ئىچىدىكى بىرىنچى role="tablist" غا باغلىنىدۇ. ھەر بىر tab نىڭ aria-controls قىممىتى شۇ قاچىدىكى ئۆزگىچە panel غا باغلانسۇن. aria-labelledby بىلەن panel نامىنى كۆرسىتىڭ. كەم دېگەندە بىر ئىشلەتكىلى بولىدىغان tab لازىم؛ تاللانغان tab بەلگىلەنمىگەن بولسا، بىرىنچى ئىشلەتكىلى بولىدىغان tab تاللىنىدۇ. select(tab, true) تاللىغاندىن كېيىن فوكۇسنىمۇ يۆتكەيدۇ؛ چەكلەنگەن ياكى باشقا گۇرۇپپىدىكى tab تاللانسا RangeError چىقىدۇ.

validateForm ۋە bindForm

const errors = validateForm(form, {
  messages: { email: 'ئېلخەت ئادرېسىنى قايتا تەكشۈرۈڭ.' },
});
// errors: [{ field, message }]

const controller = bindForm(form, {
  messages: { name: 'ئىسمىڭىزنى تولدۇرۇڭ.' },
  onValid(form, event) {
    // Your application handles this valid submission.
  },
});
controller.validate();
controller.destroy();

messages رامكىنىڭ name قىممىتىدىن خاتالىق تېكىستىگە بارىدىغان خەرىتە. ئۆز ئۇچۇرىڭىز بولمىسا ئادەتتىكى validity خاتالىقلىرىغا ماس ئۇيغۇرچە سۈكۈتتىكى ئۇچۇر ئىشلىتىلىدۇ. ئايرىم customError ياكى stepMismatch ھالىتىدە native خاتالىق تېكىستى توركۆرگۈچ تىلىدا قايتىشى مۇمكىن؛ تولۇق يەرلىكلەشتۈرۈش ئۈچۈن شۇ رامكىغا ئۆز ئۇچۇرىڭىزنى بېرىڭ. data-ug-error خاتالىق span نىڭ ID سىغا، aria-describedby ياردەم ۋە خاتالىق ID لىرىغا باغلانسۇن.

validateForm نەتىجە تىزىملىكىنى قايتۇرىدۇ ۋە رامكىلارنىڭ خاتالىق ھالىتىنى يېڭىلايدۇ. bindForm باغلانغان ۋاقىتتا novalidate نى باشقۇرۇپ، خاتا فورمىنىڭ بىرىنچى رامكىسىغا فوكۇس يۆتكەيدۇ. onValid بولسا توغرا يوللاشنى يەرلىك callback تا بىر تەرەپ قىلىدۇ؛ بولمىسا توغرا native يوللاش داۋاملىشىدۇ. destroy() ئەسلى novalidate، باشقۇرۇلغان aria-invalid ۋە خاتالىق تېكىستىنى ئەسلىگە قايتۇرىدۇ.

showToast

<div id="notifications" class="ug-toast-region"
  role="status" aria-live="polite" aria-relevant="additions text"></div>
const toast = showToast(
  document.getElementById('notifications'),
  'مەزمۇن ساقلىنىشقا تەييار.',
  { duration: 0, dismissLabel: 'ئۇقتۇرۇشنى تاقاش' },
);
// toast.element is the created element.
// toast.dismiss() and toast.destroy() both remove it.

ئۇقتۇرۇش مەزمۇنى HTML ئەمەس، ئاددىي تېكىست سۈپىتىدە قويۇلىدۇ. سۈكۈتتىكى ۋاقىت 8000 ms؛ duration: 0 بولسا ئادەم تاقىغۇچە قالىدۇ. مۇھىم نەتىجە ياكى خاتالىق ئۈچۈن ۋاقىت چەكلىمىسىز ئۇچۇرنى تاللاڭ.

SVG ۋە ئىزدەش API

نامنەتىجە
iconSVG(name, { size, strokeWidth, label, variant })SVG تېكىستى. چوڭلۇق 24، strokeWidth بولسا 1.7؛ سۈكۈتتىكى variant بولسا outline.
flagSVG({ width, label })كەڭلىك/ئېگىزلىك نىسبىتى 3:2 بولغان بايراق SVG تېكىستى.
mapSVG({ width, variant, label, color, strokeWidth })شەرقىي تۈركىستان خەرىتىسى؛ outline ياكى filled. كەڭلىك 640، رەڭ currentColor، سىزىق قېلىنلىقى 4.
mapMetadataمەنبە، ئىجازەتنامە، جۇغراپىيەلىك دائىرە ۋە projection ئۇچۇرى؛ چوڭقۇر توڭلىتىلغان.
findIcons(query, { category })سۆز ۋە تۈرگە ماس سىنبەلگە ناملىرىدىن يېڭى تىزىملىك.
iconNames · coloredIconNames · directionalIcons110 سىنبەلگە نامى، رەڭلىك نۇسخىسى بار 63 نام ۋە ئوق تاللاشقا ياردەمچى ناملار.
iconCatalog · iconCategoriesئىككى تىللىق metadata؛ ھەر سىنبەلگىنىڭ variants مەيدانى بار نۇسخىلارنى كۆرسىتىدۇ.

سىنبەلگە، بايراق ۋە ئىزدەش export لىرى يىلتىز import ۋە @uyghurcommunity/ui/icons دا بار. خەرىتە API سى يىلتىز import ۋە @uyghurcommunity/ui/maps دا بار. escapeMarkup(value) پەقەت icons تارماق import ىدا بار؛ ئۇ markup ئۈچۈن بەلگىلەرنى escape قىلىدۇ، URL ياكى CSS تەكشۈرگۈچ ئەمەس. تەپسىلاتنى سىنبەلگە ۋە خەرىتە قوللانمىسىدىن كۆرۈڭ.

پارامېتىر ۋە خاتالىق

يوق ياكى ماس كەلمەيدىغان DOM ئېلېمېنتى ئۈچۈن TypeError چىقىشى مۇمكىن. سىنبەلگە نامى، نۇسخا ۋە تۈر نامى allowlist بويىچە تەكشۈرۈلىدۇ. icon چوڭلۇقى پۈتۈن سان 12–256، strokeWidth بولسا 0.5–4؛ flag كەڭلىكى پۈتۈن سان 24–1800 بولسۇن. خەرىتە كەڭلىكى پۈتۈن سان 64–2048، strokeWidth بولسا 0.5–12؛ color پەقەت currentColor، #RGB ياكى #RRGGBB بولسۇن. نام بەلگىسى 160 ھەرپتىن ئاشمىسۇن. ئۆچۈرۈلگەن binding نى قايتا open/select/validate/setOpen قىلىش ئورنىغا يېڭى binding قۇرۇڭ.
